# 5.4 参考答案

`acl.op.set_model_dir` 注册单算子 OM 目录。H2D 是主机到设备，D2H 是设备到主机；异步提交后先同步，再回读。能导入 acl、能找到 OM，还不能说明输出正确，需要与参考结果比较。

下面的函数放在练习 Notebook 中，在给定输入单元格之后执行。沿用前面导入的 json、subprocess、Path、acl、NumPy，以及 `AclSession` 和常量。

## GEMM

A 为 `[32,128]`，B 为 `[128,64]`，输出为 `[32,64]`。单算子描述使用这组形状，输出由 C 缓冲区回读。

```python
def run_gemm():
    M, N, K = 32, 64, 128
    a_host, b_host = gemm_a, gemm_b
    c_host = np.zeros((M, N), dtype=np.float16)
    alpha_host = np.array([1.0], dtype=np.float16)
    beta_host = np.array([0.0], dtype=np.float16)
    MODEL_DIR = Path("work/05.04_chapter_test/gemm_models").resolve()
    MODEL_DIR.mkdir(parents=True, exist_ok=True)
    gemm_singleop = [{
        "op": "GEMM",
        "input_desc": [
            {"format": "ND", "shape": [M, K], "type": "float16"},  # a
            {"format": "ND", "shape": [K, N], "type": "float16"},  # b
            {"format": "ND", "shape": [M, N], "type": "float16"},  # c（输出就地写入）
            {"format": "ND", "shape": [], "type": "float16"},      # alpha：标量
            {"format": "ND", "shape": [], "type": "float16"},      # beta：标量
        ],
        "output_desc": [{"format": "ND", "shape": [M, N], "type": "float16"}],
        "attr": [
            {"name": "transpose_a", "type": "bool", "value": False},
            {"name": "transpose_b", "type": "bool", "value": False},
        ],
    }]

    json_path = MODEL_DIR / "gemm_singleop.json"
    json_path.write_text(json.dumps(gemm_singleop, indent=2, ensure_ascii=False), encoding="utf-8")
    print("已写入", json_path)

    subprocess.run([
        "atc", f"--singleop={json_path}", f"--output={MODEL_DIR}",
        f"--soc_version={SOC_VERSION}",
    ], check=True)

    with AclSession(MODEL_DIR, DEVICE_ID) as session:
        dev_a = session.to_device(a_host)
        dev_b = session.to_device(b_host)
        dev_c = session.to_device(c_host)
        dev_alpha = session.to_device(alpha_host)
        dev_beta = session.to_device(beta_host)

        # 重复算子调用，Event 只统计这一段设备执行时间。
        check(acl.rt.record_event(session.start_event, session.stream), "acl.rt.record_event start")
        for _ in range(REPEAT):
            check(acl.blas.gemm_ex(
                0, 0, 0,
                M, N, K,
                dev_alpha, dev_a, -1, ACL_FLOAT16,
                dev_b, -1, ACL_FLOAT16,
                dev_beta, dev_c, -1, ACL_FLOAT16,
                ACL_COMPUTE_HIGH_PRECISION, session.stream,
            ), "acl.blas.gemm_ex")
        check(acl.rt.record_event(session.end_event, session.stream), "acl.rt.record_event end")
        check(acl.rt.synchronize_stream(session.stream), "acl.rt.synchronize_stream")
        elapsed_ms, ret = acl.rt.event_elapsed_time(session.start_event, session.end_event)
        check(ret, "acl.rt.event_elapsed_time")

        out_host = np.empty((M, N), dtype=np.float16)
        host_ptr, ret = acl.rt.malloc_host(out_host.nbytes)
        check(ret, "acl.rt.malloc_host")
        try:
            check(acl.rt.memcpy(
                host_ptr, out_host.nbytes,
                dev_c, c_host.nbytes, MEMCPY_D2H,
            ), "acl.rt.memcpy D2H")
            result_bytes = acl.util.ptr_to_bytes(host_ptr, out_host.nbytes)
        finally:
            acl.rt.free_host(host_ptr)
        out_host = np.frombuffer(result_bytes, dtype=out_host.dtype).reshape(out_host.shape)

    np.testing.assert_allclose(out_host, gemm_golden, atol=0.02, rtol=0)
    error = np.max(np.abs(out_host.astype(np.float32) - gemm_golden.astype(np.float32)))
    print("GEMM：", out_host.shape, "平均执行 ms：", elapsed_ms / REPEAT, "最大误差：", error)
    return out_host

```

## COO SpMV

题目给定的 spmv_x 是一维 `[128]`，先 reshape 为接口所需的 `[128,1]`。矩阵为 `[64,128]`，输出为 `[64,1]`。两种算子分别使用模型目录，避免混用描述文件。

```python
def run_spmv():
    M, N, NNZ = SM, SN, len(spmv_values)
    indices, values = spmv_indices, spmv_values
    dense_shape = np.array([M, N], dtype=np.int64)
    x2 = spmv_x.reshape(N, 1)
    MODEL_DIR = Path("work/05.04_chapter_test/spmv_models").resolve()
    MODEL_DIR.mkdir(parents=True, exist_ok=True)
    spmv_singleop = [{
        "op": "SparseTensorDenseMatMul",
        "input_desc": [
            {"format": "ND", "shape": [NNZ, 2], "type": "int64"},  # indices
            {"format": "ND", "shape": [NNZ], "type": "float"},     # values
            {"format": "ND", "shape": [2], "type": "int64"},       # dense_shape
            {"format": "ND", "shape": [N, 1], "type": "float"},    # x2
        ],
        "output_desc": [{"format": "ND", "shape": [M, 1], "type": "float"}],
        "attr": [
            {"name": "adjoint_a", "type": "bool", "value": False},
            {"name": "adjoint_b", "type": "bool", "value": False},
        ],
    }]

    json_path = MODEL_DIR / "spmv_singleop.json"
    json_path.write_text(json.dumps(spmv_singleop, indent=2, ensure_ascii=False), encoding="utf-8")
    print("已写入", json_path)

    subprocess.run([
        "atc", f"--singleop={json_path}", f"--output={MODEL_DIR}",
        f"--soc_version={SOC_VERSION}",
    ], check=True)

    with AclSession(MODEL_DIR, DEVICE_ID) as session:
        idx_desc, idx_buffer = session.input_tensor(indices, ACL_INT64)
        val_desc, val_buffer = session.input_tensor(values, ACL_FLOAT)
        shape_desc, shape_buffer = session.input_tensor(dense_shape, ACL_INT64)
        x_desc, x_buffer = session.input_tensor(x2, ACL_FLOAT)

        out_host = np.empty((M, 1), dtype=np.float32)
        out_ptr, out_desc, out_buffer = session.output_tensor(out_host, ACL_FLOAT)
        attr = session.create_attr()
        check(acl.op.set_attr_bool(attr, "adjoint_a", False), "set adjoint_a")
        check(acl.op.set_attr_bool(attr, "adjoint_b", False), "set adjoint_b")

        input_descs = [idx_desc, val_desc, shape_desc, x_desc]
        input_buffers = [idx_buffer, val_buffer, shape_buffer, x_buffer]
        # 重复算子调用，Event 只统计这一段设备执行时间。
        check(acl.rt.record_event(session.start_event, session.stream), "acl.rt.record_event start")
        for _ in range(REPEAT):
            check(acl.op.execute_v2(
                "SparseTensorDenseMatMul",
                input_descs, input_buffers,
                [out_desc], [out_buffer], attr, session.stream,
            ), "acl.op.execute_v2")
        check(acl.rt.record_event(session.end_event, session.stream), "acl.rt.record_event end")
        check(acl.rt.synchronize_stream(session.stream), "acl.rt.synchronize_stream")
        elapsed_ms, ret = acl.rt.event_elapsed_time(session.start_event, session.end_event)
        check(ret, "acl.rt.event_elapsed_time")
        host_ptr, ret = acl.rt.malloc_host(out_host.nbytes)
        check(ret, "acl.rt.malloc_host")
        try:
            check(acl.rt.memcpy(
                host_ptr, out_host.nbytes,
                out_ptr, out_host.nbytes, MEMCPY_D2H,
            ), "acl.rt.memcpy D2H")
            result_bytes = acl.util.ptr_to_bytes(host_ptr, out_host.nbytes)
        finally:
            acl.rt.free_host(host_ptr)
        out_host = np.frombuffer(result_bytes, dtype=out_host.dtype).reshape(out_host.shape)

    np.testing.assert_allclose(out_host[:, 0], spmv_golden, atol=1e-3, rtol=0)
    error = np.max(np.abs(out_host[:, 0] - spmv_golden))
    print("SpMV：", out_host.shape, "平均执行 ms：", elapsed_ms / REPEAT, "最大误差：", error)
    return out_host

```

## 调用并核对

```python
gemm_output = run_gemm()
spmv_output = run_spmv()
```

两次返回的都是已回读的 NumPy 数组，退出 AclSession 后仍可使用。GEMM 使用 `atol=0.02`，SpMV 使用 `atol=1e-3`，两者 rtol=0。SpMV 在本实验中走 AI CPU/tf_kernel。时间填写当前运行值。
